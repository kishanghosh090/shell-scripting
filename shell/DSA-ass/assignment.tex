\section*{Program 1: Binary Tree Implementation}
\addcontentsline{toc}{section}{Program 1: Binary Tree Implementation}

\subsection*{Problem Statement}
Write a menu-based C++ program to implement a Binary Tree. The program should support the following operations: Insert, Delete, Search, Height, In-order Traversal, Pre-order Traversal, Post-order Traversal, Level Order Traversal, and Exit.

\subsection*{Algorithm}
\begin{enumerate}
    \item Create a Node class with data, left and right pointers.
    \item Create a BinaryTree class that uses level-order insertion (complete binary tree).
    \item \textbf{Insert}: Use a queue to find the first empty position and insert the new node.
    \item \textbf{Delete}: Find the target node and the deepest node. Replace target's data with deepest node's data, then delete the deepest node.
    \item \textbf{Search}: Level-order traversal to find the key.
    \item \textbf{Height}: Recursively compute max(left subtree height, right subtree height) + 1.
    \item \textbf{Traversals}: Implement recursive in-order, pre-order, post-order, and iterative level-order using a queue.
    \item Display a menu and handle user choice using a switch statement.
\end{enumerate}

\subsection*{Source Code (01.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
#include <queue>
using namespace std;

class Node
{
public:
    int data;
    Node *left, *right;

    Node(int val)
    {
        data = val;
        left = right = NULL;
    }
};

class BinaryTree
{
private:
    Node *root;

    int height(Node *node)
    {
        if (node == NULL) return -1;
        return 1 + max(height(node->left), height(node->right));
    }

    void inorder(Node *node)
    {
        if (node)
        {
            inorder(node->left);
            cout << node->data << " ";
            inorder(node->right);
        }
    }

    void preorder(Node *node)
    {
        if (node)
        {
            cout << node->data << " ";
            preorder(node->left);
            preorder(node->right);
        }
    }

    void postorder(Node *node)
    {
        if (node)
        {
            postorder(node->left);
            postorder(node->right);
            cout << node->data << " ";
        }
    }

public:
    BinaryTree() { root = NULL; }

    void insert(int value)
    {
        Node *newNode = new Node(value);
        if (root == NULL) { root = newNode; return; }

        queue<Node *> q;
        q.push(root);

        while (!q.empty())
        {
            Node *temp = q.front(); q.pop();
            if (temp->left == NULL)
            { temp->left = newNode; return; }
            else q.push(temp->left);

            if (temp->right == NULL)
            { temp->right = newNode; return; }
            else q.push(temp->right);
        }
    }

    bool search(int key)
    {
        if (root == NULL) return false;
        queue<Node *> q; q.push(root);
        while (!q.empty())
        {
            Node *temp = q.front(); q.pop();
            if (temp->data == key) return true;
            if (temp->left) q.push(temp->left);
            if (temp->right) q.push(temp->right);
        }
        return false;
    }

    void deleteNode(int key)
    {
        if (root == NULL) return;
        if (root->left == NULL && root->right == NULL)
        {
            if (root->data == key) { delete root; root = NULL; }
            return;
        }

        queue<Node *> q; q.push(root);
        Node *target = NULL, *temp;

        while (!q.empty())
        {
            temp = q.front(); q.pop();
            if (temp->data == key) target = temp;
            if (temp->left) q.push(temp->left);
            if (temp->right) q.push(temp->right);
        }

        if (target != NULL)
        {
            int deepestValue = temp->data;
            queue<Node *> q2; q2.push(root);
            while (!q2.empty())
            {
                Node *curr = q2.front(); q2.pop();
                if (curr->left)
                {
                    if (curr->left == temp)
                    { delete curr->left; curr->left = NULL; break; }
                    q2.push(curr->left);
                }
                if (curr->right)
                {
                    if (curr->right == temp)
                    { delete curr->right; curr->right = NULL; break; }
                    q2.push(curr->right);
                }
            }
            target->data = deepestValue;
        }
    }

    void displayHeight()
    { cout << "Height = " << height(root) << endl; }

    void inorderTraversal()
    { cout << "Inorder: "; inorder(root); cout << endl; }

    void preorderTraversal()
    { cout << "Preorder: "; preorder(root); cout << endl; }

    void postorderTraversal()
    { cout << "Postorder: "; postorder(root); cout << endl; }

    void levelOrderTraversal()
    {
        if (root == NULL) { cout << "Tree is empty\n"; return; }
        queue<Node *> q; q.push(root);
        cout << "Level Order: ";
        while (!q.empty())
        {
            Node *temp = q.front(); q.pop();
            cout << temp->data << " ";
            if (temp->left) q.push(temp->left);
            if (temp->right) q.push(temp->right);
        }
        cout << endl;
    }
};

int main()
{
    BinaryTree tree;
    int choice, value;
    do
    {
        cout << "\n===== MENU =====\n";
        cout << "1. Insert\n2. Delete\n3. Search\n4. Height\n";
        cout << "5. Inorder Traversal\n6. Preorder Traversal\n";
        cout << "7. Postorder Traversal\n8. Level Order Traversal\n";
        cout << "9. Exit\nEnter choice: ";
        cin >> choice;

        switch (choice)
        {
        case 1: cout << "Enter value: "; cin >> value; tree.insert(value); break;
        case 2: cout << "Enter value to delete: "; cin >> value; tree.deleteNode(value); break;
        case 3: cout << "Enter value to search: "; cin >> value;
            if (tree.search(value)) cout << "Found\n"; else cout << "Not Found\n"; break;
        case 4: tree.displayHeight(); break;
        case 5: tree.inorderTraversal(); break;
        case 6: tree.preorderTraversal(); break;
        case 7: tree.postorderTraversal(); break;
        case 8: tree.levelOrderTraversal(); break;
        case 9: cout << "Exiting...\n"; break;
        default: cout << "Invalid choice\n";
        }
    } while (choice != 9);
    return 0;
}

\section*{Program 2: Binary Search Tree Implementation}
\addcontentsline{toc}{section}{Program 2: Binary Search Tree Implementation}

\subsection*{Problem Statement}
Write a menu-based C++ program to implement a Binary Search Tree. The program should support: Insert, Delete, Search, Height, In-order Traversal, Pre-order Traversal, Post-order Traversal, Number of Nodes, Smallest Node, and Exit.

\subsection*{Algorithm}
\begin{enumerate}
    \item Create a Node class with data, left and right pointers.
    \item \textbf{Insert}: Recursively traverse — if value $<$ node data, go left; if $>$ go right. Insert when NULL is reached.
    \item \textbf{Delete}: Three cases — (1) Leaf node: delete directly. (2) One child: replace with child. (3) Two children: find inorder successor (smallest in right subtree), replace data, delete successor.
    \item \textbf{Search}: Recursively compare with node data; go left if smaller, right if larger.
    \item \textbf{Height}: Recursively compute max(left, right) + 1.
    \item \textbf{Smallest Node}: Find the leftmost node.
    \item \textbf{Node Count}: Recursively count nodes in left + right + 1.
\end{enumerate}

\subsection*{Source Code (02.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
using namespace std;

class Node
{
public:
    int data;
    Node *left, *right;
    Node(int val) { data = val; left = right = NULL; }
};

class BST
{
private:
    Node *root;

    Node *insert(Node *node, int val)
    {
        if (node == NULL) return new Node(val);
        if (val < node->data) node->left = insert(node->left, val);
        else if (val > node->data) node->right = insert(node->right, val);
        return node;
    }

    Node *findMin(Node *node)
    {
        while (node && node->left != NULL) node = node->left;
        return node;
    }

    Node *deleteNode(Node *node, int key)
    {
        if (node == NULL) return NULL;
        if (key < node->data) node->left = deleteNode(node->left, key);
        else if (key > node->data) node->right = deleteNode(node->right, key);
        else
        {
            if (node->left == NULL && node->right == NULL)
            { delete node; return NULL; }
            if (node->left == NULL)
            { Node *temp = node->right; delete node; return temp; }
            if (node->right == NULL)
            { Node *temp = node->left; delete node; return temp; }
            Node *temp = findMin(node->right);
            node->data = temp->data;
            node->right = deleteNode(node->right, temp->data);
        }
        return node;
    }

    bool search(Node *node, int key)
    {
        if (node == NULL) return false;
        if (node->data == key) return true;
        if (key < node->data) return search(node->left, key);
        return search(node->right, key);
    }

    int height(Node *node)
    {
        if (node == NULL) return -1;
        return 1 + max(height(node->left), height(node->right));
    }

    void inorder(Node *node)
    { if (node) { inorder(node->left); cout << node->data << " "; inorder(node->right); } }

    void preorder(Node *node)
    { if (node) { cout << node->data << " "; preorder(node->left); preorder(node->right); } }

    void postorder(Node *node)
    { if (node) { postorder(node->left); postorder(node->right); cout << node->data << " "; } }

    int countNodes(Node *node)
    { if (node == NULL) return 0; return 1 + countNodes(node->left) + countNodes(node->right); }

public:
    BST() { root = NULL; }
    void insert(int val) { root = insert(root, val); }
    void deleteValue(int key) { root = deleteNode(root, key); }
    void searchValue(int key)
    { if (search(root, key)) cout << "Node Found\n"; else cout << "Node Not Found\n"; }
    void printHeight() { cout << "Height = " << height(root) << endl; }
    void inorderTraversal() { cout << "Inorder: "; inorder(root); cout << endl; }
    void preorderTraversal() { cout << "Preorder: "; preorder(root); cout << endl; }
    void postorderTraversal() { cout << "Postorder: "; postorder(root); cout << endl; }
    void printNodeCount() { cout << "Total Nodes = " << countNodes(root) << endl; }
    void printSmallestNode()
    {
        if (root == NULL) { cout << "Tree is empty\n"; return; }
        Node *temp = findMin(root);
        cout << "Smallest Node = " << temp->data << endl;
    }
};

int main()
{
    BST tree; int choice, value;
    do {
        cout << "\n===== BST MENU =====\n";
        cout << "1. Insert\n2. Delete\n3. Search\n4. Height\n";
        cout << "5. Inorder Traversal\n6. Preorder Traversal\n";
        cout << "7. Postorder Traversal\n8. Number of Nodes\n";
        cout << "9. Smallest Node\n10. Exit\nEnter choice: ";
        cin >> choice;
        switch (choice)
        {
        case 1: cout << "Enter value: "; cin >> value; tree.insert(value); break;
        case 2: cout << "Enter value to delete: "; cin >> value; tree.deleteValue(value); break;
        case 3: cout << "Enter value to search: "; cin >> value; tree.searchValue(value); break;
        case 4: tree.printHeight(); break;
        case 5: tree.inorderTraversal(); break;
        case 6: tree.preorderTraversal(); break;
        case 7: tree.postorderTraversal(); break;
        case 8: tree.printNodeCount(); break;
        case 9: tree.printSmallestNode(); break;
        case 10: cout << "Exiting...\n"; break;
        default: cout << "Invalid Choice\n";
        }
    } while (choice != 10);
    return 0;
}

\section*{Program 3: Linear Search}
\addcontentsline{toc}{section}{Program 3: Linear Search}

\subsection*{Problem Statement}
Write a program to search a number from a list of given numbers using linear search. Discuss the complexity of linear search.

\subsection*{Algorithm}
\begin{enumerate}
    \item Start from the first element of the array.
    \item Compare each element with the key.
    \item If a match is found, return the index.
    \item If the end of the array is reached without a match, return -1.
\end{enumerate}

\subsection*{Complexity Analysis}
\begin{itemize}
    \item \textbf{Time Complexity}: O(n) — In the worst case, all n elements are compared.
    \item \textbf{Best Case}: O(1) — The element is found at the first position.
    \item \textbf{Worst Case}: O(n) — The element is at the last position or not present.
    \item \textbf{Space Complexity}: O(1) — No extra space is used.
\end{itemize}

\subsection*{Source Code (03.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
#include <vector>
using namespace std;

int linearSearch(const vector<int> &arr, int key)
{
    for (int i = 0; i < arr.size(); i++)
    {
        if (arr[i] == key)
            return i;
    }
    return -1;
}

int main()
{
    int n, key;
    cout << "Enter number of elements: ";
    cin >> n;

    vector<int> arr(n);
    cout << "Enter elements:\n";
    for (int i = 0; i < n; i++) cin >> arr[i];

    cout << "Enter element to search: ";
    cin >> key;

    int result = linearSearch(arr, key);
    if (result != -1)
        cout << "Element found at position " << result + 1 << endl;
    else
        cout << "Element not found." << endl;
    return 0;
}
\end{lstlisting}

\section*{Program 4: Binary Search}
\addcontentsline{toc}{section}{Program 4: Binary Search}

\subsection*{Problem Statement}
Write a program to search a number from a list of given numbers using Binary search. Discuss the complexity of Binary search.

\subsection*{Algorithm}
\begin{enumerate}
    \item The array must be sorted in ascending order.
    \item Set low = 0, high = n-1.
    \item While low $\leq$ high: Compute mid = low + (high - low) / 2.
    \item If arr[mid] == key, return mid.
    \item If arr[mid] < key, search the right half (low = mid + 1).
    \item Otherwise, search the left half (high = mid - 1).
    \item If the loop ends, return -1 (not found).
\end{enumerate}

\subsection*{Complexity Analysis}
\begin{itemize}
    \item \textbf{Time Complexity}: O(log n) — The search space is halved in each step.
    \item \textbf{Best Case}: O(1) — The element is at the middle.
    \item \textbf{Worst Case}: O(log n) — Element is at an edge or not present.
    \item \textbf{Space Complexity}: O(1) for iterative implementation.
\end{itemize}

\subsection*{Source Code (04.cpp)}

#include <iostream>
#include <vector>
using namespace std;

int binarySearch(const vector<int> &arr, int key)
{
    int low = 0;
    int high = arr.size() - 1;

    while (low <= high)
    {
        int mid = low + (high - low) / 2;

        if (arr[mid] == key) return mid;
        if (arr[mid] < key) low = mid + 1;
        else high = mid - 1;
    }
    return -1;
}

    cout << "Enter element to search: ";
    cin >> key;

    if (result != -1)
        cout << "Element found at position " << result + 1 << endl;
    else
        cout << "Element not found." << endl;

\subsection*{Problem Statement}

\subsection*{Algorithm}
\begin{enumerate}
    \item For i = 1 to n-1:
    \item   Store arr[i] as key.
    \item   Set j = i-1.
    \item   While j $\geq$ 0 and arr[j] $>$ key: shift arr[j] to arr[j+1], decrement j.
    \item   Place key at arr[j+1].
\end{enumerate}

\subsection*{Complexity Analysis}
\begin{itemize}
    \item \textbf{Time Complexity}: O(n$^2$) in worst and average cases.
    \item \textbf{Best Case}: O(n) — Array is already sorted.
    \item \textbf{Space Complexity}: O(1) — In-place sorting.
    \item \textbf{Stable}: Yes, relative order of equal elements is preserved.
\end{itemize}

\subsection*{Source Code (06.cpp)}

void insertionSort(vector<int> &arr)
{
    int n = arr.size();

    {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key)
        {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;
    }
}

\section*{Program 8: Quick Sort (Descending)}
\addcontentsline{toc}{section}{Program 8: Quick Sort (Descending)}

\subsection*{Problem Statement}
Write a program to sort a list of elements in descending order using Quick Sort. Discuss the complexity of this algorithm.

\subsection*{Algorithm}
\begin{enumerate}
    \item Choose the last element as pivot.
    \item Partition: Arrange elements so that all elements greater than pivot are on the left, and smaller on the right (for descending order).
    \item Recursively apply Quick Sort to the left and right subarrays.
\end{enumerate}

\subsection*{Complexity Analysis}
\begin{itemize}
    \item \textbf{Time Complexity (Average)}: O(n log n).
    \item \textbf{Worst Case}: O(n$^2$) — When pivot is always the smallest or largest element.
    \item \textbf{Best Case}: O(n log n) — When pivot divides the array into equal halves.
    \item \textbf{Space Complexity}: O(log n) — Due to recursive stack.
\end{itemize}

\subsection*{Source Code (08.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
#include <vector>
using namespace std;

int partition(vector<int> &arr, int low, int high)
{
    int pivot = arr[high];
    int i = low - 1;

    for (int j = low; j < high; j++)
    {
        if (arr[j] > pivot)
        {
            i++;
            swap(arr[i], arr[j]);
        }
    }
    swap(arr[i + 1], arr[high]);
    return i + 1;
}

void quickSort(vector<int> &arr, int low, int high)
{
    if (low < high)
    {
        int pi = partition(arr, low, high);
        quickSort(arr, low, pi - 1);
        quickSort(arr, pi + 1, high);
    }
}

int main()
{
    int n;
    cout << "Enter number of elements: ";
    cin >> n;

    vector<int> arr(n);
    cout << "Enter elements:\n";
    for (int i = 0; i < n; i++) cin >> arr[i];

    quickSort(arr, 0, n - 1);

    cout << "Sorted Array (Descending): ";
    for (int x : arr) cout << x << " ";
    cout << endl;
    return 0;
}

\subsection*{Problem Statement}

\subsection*{Algorithm}

    \item Merge the sorted halves by comparing elements and combining them in sorted order.

\subsection*{Complexity Analysis}

#include <vector>
using namespace std;

void merge(vector<int> &arr, int left, int mid, int right)
{
    vector<int> temp;
    int i = left, j = mid + 1;

    while (i <= mid && j <= right)
    {
        if (arr[i] <= arr[j])
        { temp.push_back(arr[i]); i++; }
        else
        { temp.push_back(arr[j]); j++; }
    }
    while (i <= mid) { temp.push_back(arr[i]); i++; }
    while (j <= right) { temp.push_back(arr[j]); j++; }

    for (int k = 0; k < temp.size(); k++)
        arr[left + k] = temp[k];
}

{
    if (left < right)
    {
        int mid = left + (right - left) / 2;
        mergeSort(arr, left, mid);
        mergeSort(arr, mid + 1, right);
        merge(arr, left, mid, right);
    }
}

\subsection*{Problem Statement}
Write a program to check if a binary tree is height-balanced. A binary tree is height-balanced if for every node, the difference between the heights of its left and right subtrees is at most 1.

\subsection*{Algorithm}
\begin{enumerate}
    \item Recursively compute the height of left and right subtrees.
    \item If any subtree returns -1, propagate -1 upward (unbalanced).
    \item If |leftHeight - rightHeight| $>$ 1, return -1 (unbalanced).
    \item Otherwise, return max(leftHeight, rightHeight) + 1.
\end{enumerate}

\subsection*{Source Code (11.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
#include <algorithm>
using namespace std;

class Node
{
public:
    int data;
    Node *left, *right;
    Node(int val) { data = val; left = right = NULL; }
};

int checkBalance(Node *root)
{
    if (root == NULL) return 0;

    int leftHeight = checkBalance(root->left);
    if (leftHeight == -1) return -1;

    int rightHeight = checkBalance(root->right);
    if (rightHeight == -1) return -1;

    if (abs(leftHeight - rightHeight) > 1) return -1;

    return max(leftHeight, rightHeight) + 1;
}

bool isBalanced(Node *root)
{ return checkBalance(root) != -1; }

int main()
{
    Node *root = new Node(1);
    root->left = new Node(2);
    root->right = new Node(3);
    root->left->left = new Node(4);
    root->left->right = new Node(5);

    if (isBalanced(root))
        cout << "Binary Tree is Height Balanced" << endl;
    else
        cout << "Binary Tree is NOT Height Balanced" << endl;

    return 0;
}

\subsection*{Problem Statement}
Write a program to implement a priority queue using a linked list. Include operations such as enqueue with priority, dequeue, and display the elements.

\subsection*{Algorithm}
\begin{enumerate}
    \item Enqueue: Insert the new node such that the list remains sorted by priority (lower number = higher priority).
    \item Dequeue: Remove and return the front node (highest priority).
    \item Display: Traverse the list showing element data and priority.
\end{enumerate}

\subsection*{Source Code (12.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
using namespace std;

class Node {
public:
    int data, priority;
    Node* next;
    Node(int d, int p) { data = d; priority = p; next = NULL; }
};

class PriorityQueue {
private:
    Node* front;
public:
    PriorityQueue() { front = NULL; }

    void enqueue(int data, int priority) {
        Node* newNode = new Node(data, priority);
        if (front == NULL || priority < front->priority) {
            newNode->next = front; front = newNode; return;
        }
        Node* temp = front;
        while (temp->next != NULL && temp->next->priority <= priority)
            temp = temp->next;
        newNode->next = temp->next; temp->next = newNode;
    }

    void dequeue() {
        if (front == NULL) { cout << "Priority Queue Underflow\n"; return; }
        Node* temp = front;
        cout << "Deleted Element: " << front->data << endl;
        front = front->next; delete temp;
    }

    void display() {
        if (front == NULL) { cout << "Priority Queue is Empty\n"; return; }
        Node* temp = front;
        cout << "\nElement\tPriority\n";
        while (temp != NULL) {
            cout << temp->data << "\t" << temp->priority << endl;
            temp = temp->next;
        }
    }
};

                cout << "Enter priority: "; cin >> priority;

        case 3: pq.display(); break;

\section*{Program 16: Infix to Postfix Conversion}
\addcontentsline{toc}{section}{Program 16: Infix to Postfix}

\subsection*{Problem Statement}
Write a program to convert an infix expression to postfix notation using a stack.

\subsection*{Algorithm}
\begin{enumerate}
    \item Scan the infix expression from left to right.
    \item If operand, add to postfix string.
    \item If '(', push to stack.
    \item If ')', pop from stack until '(' is found.
    \item If operator, pop operators with higher or equal precedence, then push current operator.
    \item After scanning, pop all remaining operators.
\end{enumerate}

\subsection*{Source Code (16.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
#include <stack>
#include <string>
using namespace std;

int precedence(char op) {
    if (op == '^') return 3;
    if (op == '*' || op == '/') return 2;
    if (op == '+' || op == '-') return 1;
    return 0;
}

bool isOperand(char ch) { return isalnum(ch); }

string infixToPostfix(string infix) {
    stack<char> st;
    string postfix = "";
    for (char ch : infix) {
        if (isOperand(ch)) { postfix += ch; }
        else if (ch == '(') { st.push(ch); }
        else if (ch == ')') {
            while (!st.empty() && st.top() != '(') {
                postfix += st.top(); st.pop();
            }
            if (!st.empty()) st.pop();
        }
        else {
            while (!st.empty() && precedence(st.top()) >= precedence(ch)) {
                postfix += st.top(); st.pop();
            }
            st.push(ch);
        }
    }
    while (!st.empty()) { postfix += st.top(); st.pop(); }
    return postfix;
}

int main() {
    string infix;
    cout << "Enter Infix Expression: "; cin >> infix;
    cout << "Postfix Expression: " << infixToPostfix(infix) << endl;
    return 0;
}

\section*{Program 22: Circular Linked List}

\subsection*{Problem Statement}
Implement Circular Linked List. Include functions for insertion, deletion, searching, and display.

\subsection*{Algorithm}
\begin{enumerate}
    \item Create a Node class with data and next pointer.
    \item Maintain a last pointer initialized to NULL (last node points to first node).
    \item \textbf{Insert}: Create a new node. If list is empty, set last to new node and point its next to itself. Otherwise, insert after last and update last.
    \item \textbf{Delete}: Traverse from last->next (first node). Find the node with the value and remove it by adjusting the previous node's next pointer.
    \item \textbf{Search}: Traverse the circular list using a do-while loop until we return to the start.
    \item \textbf{Display}: Traverse the circular list and print each node's data, ending with "(back to start)".
\end{enumerate}

using namespace std;

public:
    int data;
    Node *next;
    Node(int val) { data = val; next = NULL; }
};

class CircularLinkedList {
private:
    Node *last;
public:
    CircularLinkedList() { last = NULL; }

    void insert(int value) {
        Node *newNode = new Node(value);
        if (last == NULL) { last = newNode; last->next = last; return; }
        newNode->next = last->next; last->next = newNode; last = newNode;
    }

        if (last == NULL) { cout << "List is empty\n"; return; }
        Node *curr = last->next, *prev = last;
        do {
            if (curr->data == value) {
                if (curr == last && curr == last->next) { delete curr; last = NULL; }
                else if (curr == last) { prev->next = curr->next; last = prev; delete curr; }
                else { prev->next = curr->next; delete curr; }
                cout << "Node deleted\n"; return;
            }
            prev = curr; curr = curr->next;
        } while (curr != last->next);
        cout << "Value not found\n";

        if (last == NULL) { cout << "List is empty\n"; return; }
        Node *temp = last->next; int pos = 1;
        do {
            if (temp->data == value) { cout << "Element found at position " << pos << endl; return; }
            temp = temp->next; pos++;
        } while (temp != last->next);
        cout << "Element not found\n";
    }

    void display() {
        if (last == NULL) { cout << "List is empty\n"; return; }
        Node *temp = last->next;
        cout << "Circular List: ";
        do { cout << temp->data << " "; temp = temp->next; } while (temp != last->next);
        cout << "(back to start)\n";
    }

\section*{Program 23: Reverse a Linked List}
\addcontentsline{toc}{section}{Program 23: Reverse a Linked List}

\subsection*{Problem Statement}
Write a program to reverse a Linked List.

\subsection*{Algorithm}
\begin{enumerate}
    \item Initialize three pointers: prev = NULL, curr = head, nextNode = NULL.
    \item While curr != NULL: Store curr->next in nextNode. Set curr->next = prev. Move prev to curr. Move curr to nextNode.
    \item Set head = prev.
\end{enumerate}

\subsection*{Source Code (23.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
using namespace std;

class Node {
public:
    int data;
    Node *next;
    Node(int val) { data = val; next = NULL; }
};

class LinkedList {
private:
    Node *head;
public:
    LinkedList() { head = NULL; }

    void insert(int value) {
        Node *newNode = new Node(value);
        if (head == NULL) { head = newNode; return; }
        Node *temp = head;
        while (temp->next != NULL) temp = temp->next;
        temp->next = newNode;
    }

    void reverse() {
        Node *prev = NULL, *curr = head, *nextNode = NULL;
        while (curr != NULL) {
            nextNode = curr->next;
            curr->next = prev;
            prev = curr;
            curr = nextNode;
        }
        head = prev;
    }

    void display() {
        Node *temp = head;
        while (temp != NULL) { cout << temp->data << " -> "; temp = temp->next; }
        cout << "NULL" << endl;
    }
};

int main() {
    LinkedList list;
    int n, value;
    cout << "Enter number of nodes: "; cin >> n;
    cout << "Enter elements:\n";
    for (int i = 0; i < n; i++) { cin >> value; list.insert(value); }

    cout << "\nOriginal Linked List:\n"; list.display();
    list.reverse();
    cout << "\nReversed Linked List:\n"; list.display();
    return 0;
}
\end{lstlisting}

\section*{Program 24: Stack using Linked List}
\addcontentsline{toc}{section}{Program 24: Stack using Linked List}

\subsection*{Problem Statement}
Write a program to perform Stack operations using Linked List implementation.

\subsection*{Algorithm}
\begin{enumerate}
    \item Create a Node class with data and next pointer.
    \item Maintain a top pointer initialized to NULL.
    \item \textbf{Push}: Create a new node, set its next to current top, and update top to the new node.
    \item \textbf{Pop}: If top is NULL, display "Stack Underflow". Otherwise, remove the top node and update top to the next node.
    \item \textbf{Peek}: Return top's data without removing it.
    \item \textbf{Display}: Traverse from top to NULL, printing each node's data.
\end{enumerate}

\subsection*{Source Code (24.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
using namespace std;

class Node {
public:
    int data;
    Node *next;
    Node(int val) { data = val; next = NULL; }
};

class Stack {
private:
    Node *top;
public:
    Stack() { top = NULL; }

    void push(int value) {
        Node *newNode = new Node(value);
        newNode->next = top; top = newNode;
        cout << value << " pushed into stack.\n";
    }

    void pop() {
        if (top == NULL) { cout << "Stack Underflow\n"; return; }
        Node *temp = top;
        cout << "Popped Element: " << top->data << endl;
        top = top->next; delete temp;
    }

    void peek() {
        if (top == NULL) { cout << "Stack is Empty\n"; return; }
        cout << "Top Element: " << top->data << endl;
    }

    void display() {
        if (top == NULL) { cout << "Stack is Empty\n"; return; }
        Node *temp = top;
        cout << "Stack Elements (Top to Bottom): ";
        while (temp != NULL) { cout << temp->data << " "; temp = temp->next; }
        cout << endl;
    }
};

int main() {
    Stack s; int choice, value;
    do {
        cout << "\n===== STACK MENU =====\n";
        cout << "1. Push\n2. Pop\n3. Peek\n4. Display\n5. Exit\nEnter Choice: ";
        cin >> choice;
        switch (choice) {
        case 1: cout << "Enter value: "; cin >> value; s.push(value); break;
        case 2: s.pop(); break;
        case 3: s.peek(); break;
        case 4: s.display(); break;
        case 5: cout << "Exiting...\n"; break;
        default: cout << "Invalid Choice\n";
        }
    } while (choice != 5);
    return 0;
}

\section*{Program 25: Stack using Array}
\addcontentsline{toc}{section}{Program 25: Stack using Array}

\subsection*{Problem Statement}
Perform Stack operations using Array implementation.

\subsection*{Algorithm}
\begin{enumerate}
    \item Use an array of fixed size (e.g., 100) and a top variable initialized to -1.
    \item \textbf{Push}: If top == size-1, display "Stack Overflow". Otherwise, increment top and insert value at arr[top].
    \item \textbf{Pop}: If top == -1, display "Stack Underflow". Otherwise, return arr[top] and decrement top.
    \item \textbf{Peek}: If top == -1, display "Stack is Empty". Otherwise, return arr[top].
    \item \textbf{Display}: Traverse from top down to 0 and print each element.
\end{enumerate}

\subsection*{Source Code (25.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
using namespace std;

class Stack {
private:
    int arr[100], top;
public:
    Stack() { top = -1; }

    void push(int value) {
        if (top == 99) { cout << "Stack Overflow\n"; return; }
        arr[++top] = value;
        cout << value << " pushed into stack.\n";
    }

    void pop() {
        if (top == -1) { cout << "Stack Underflow\n"; return; }
        cout << "Popped Element: " << arr[top--] << endl;
    }

    void peek() {
        if (top == -1) { cout << "Stack is Empty\n"; return; }
        cout << "Top Element: " << arr[top] << endl;
    }

    void display() {
        if (top == -1) { cout << "Stack is Empty\n"; return; }
        cout << "Stack Elements (Top to Bottom): ";
        for (int i = top; i >= 0; i--) cout << arr[i] << " ";
        cout << endl;
    }
};

int main() {
    Stack s; int choice, value;
    do {
        cout << "\n===== STACK MENU =====\n";
        cout << "1. Push\n2. Pop\n3. Peek\n4. Display\n5. Exit\nEnter Choice: ";
        cin >> choice;
        switch (choice) {
        case 1: cout << "Enter value: "; cin >> value; s.push(value); break;
        case 2: s.pop(); break;
        case 3: s.peek(); break;
        case 4: s.display(); break;
        case 5: cout << "Exiting...\n"; break;
        default: cout << "Invalid Choice\n";
        }
    } while (choice != 5);
    return 0;
}

\section*{Program 26: Double-ended Queue using Linked List}
\addcontentsline{toc}{section}{Program 26: Deque using Linked List}

\subsection*{Problem Statement}
Write a program to create and perform different operations on Double-ended Queues using Linked List implementation.

\subsection*{Algorithm}
\begin{enumerate}
    \item Create a Node class with data, prev, and next pointers.
    \item Maintain front and rear pointers initialized to NULL.
    \item \textbf{Insert Front}: Create a new node. If deque is empty, set front = rear = newNode. Otherwise, link new node before front and update front.
    \item \textbf{Insert Rear}: Create a new node. If deque is empty, set front = rear = newNode. Otherwise, link new node after rear and update rear.
    \item \textbf{Delete Front}: If deque is empty, display "Underflow". Remove front node and update front pointer.
    \item \textbf{Delete Rear}: If deque is empty, display "Underflow". Remove rear node and update rear pointer.
    \item \textbf{Display}: Traverse from front to rear and print each element.
\end{enumerate}

\subsection*{Source Code (26.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
using namespace std;

class Node {
public:
    int data;
    Node *prev, *next;
    Node(int val) { data = val; prev = NULL; next = NULL; }
};

class Deque {
private:
    Node *front, *rear;
public:
    Deque() { front = rear = NULL; }

    void insertFront(int value) {
        Node *newNode = new Node(value);
        if (front == NULL) { front = rear = newNode; }
        else { newNode->next = front; front->prev = newNode; front = newNode; }
        cout << value << " inserted at front.\n";
    }

    void insertRear(int value) {
        Node *newNode = new Node(value);
        if (rear == NULL) { front = rear = newNode; }
        else { rear->next = newNode; newNode->prev = rear; rear = newNode; }
        cout << value << " inserted at rear.\n";
    }

    void deleteFront() {
        if (front == NULL) { cout << "Deque Underflow\n"; return; }
        Node *temp = front;
        cout << "Deleted from front: " << front->data << endl;
        if (front == rear) { front = rear = NULL; }
        else { front = front->next; front->prev = NULL; }
        delete temp;
    }

    void deleteRear() {
        if (rear == NULL) { cout << "Deque Underflow\n"; return; }
        Node *temp = rear;
        cout << "Deleted from rear: " << rear->data << endl;
        if (front == rear) { front = rear = NULL; }
        else { rear = rear->prev; rear->next = NULL; }
        delete temp;
    }

    void display() {
        if (front == NULL) { cout << "Deque is Empty\n"; return; }
        Node *temp = front;
        cout << "Deque Elements: ";
        while (temp != NULL) { cout << temp->data << " "; temp = temp->next; }
        cout << endl;
    }
};

int main() {
    Deque dq; int choice, value;
    do {
        cout << "\n===== DEQUE MENU =====\n";
        cout << "1. Insert Front\n2. Insert Rear\n";
        cout << "3. Delete Front\n4. Delete Rear\n5. Display\n6. Exit\nEnter Choice: ";
        cin >> choice;
        switch (choice) {
        case 1: cout << "Enter value: "; cin >> value; dq.insertFront(value); break;
        case 2: cout << "Enter value: "; cin >> value; dq.insertRear(value); break;
        case 3: dq.deleteFront(); break;
        case 4: dq.deleteRear(); break;
        case 5: dq.display(); break;
        case 6: cout << "Exiting...\n"; break;
        default: cout << "Invalid Choice\n";
        }
    } while (choice != 6);
    return 0;
}

\section*{Program 27: Polynomial Addition using Linked List}
\addcontentsline{toc}{section}{Program 27: Polynomial Addition}

\subsection*{Problem Statement}
Write a program to represent a polynomial using linked list and add two polynomials.

\subsection*{Algorithm}
\begin{enumerate}
    \item Create a Node class with coefficient, exponent, and next pointer.
    \item Insert terms with (coefficient, exponent) in descending order of exponents.
    \item To add two polynomials: Traverse both lists simultaneously.
    \item If exponents are equal, add coefficients and insert the result.
    \item If one exponent is larger, insert that term and move its pointer.
    \item Continue until both lists are exhausted.
\end{enumerate}

\subsection*{Source Code (27.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
using namespace std;

class Term {
public:
    int coeff, exp;
    Term *next;
    Term(int c, int e) { coeff = c; exp = e; next = NULL; }
};

class Polynomial {
private:
    Term *head;
public:
    Polynomial() { head = NULL; }

    void insertTerm(int coeff, int exp) {
        Term *newTerm = new Term(coeff, exp);
        if (head == NULL || head->exp < exp) {
            newTerm->next = head; head = newTerm; return;
        }
        Term *temp = head;
        while (temp->next != NULL && temp->next->exp >= exp)
            temp = temp->next;
        if (temp->exp == exp) { temp->coeff += coeff; delete newTerm; return; }
        newTerm->next = temp->next; temp->next = newTerm;
    }

    void display() {
        Term *temp = head;
        while (temp != NULL) {
            cout << temp->coeff << "x^" << temp->exp;
            if (temp->next != NULL) cout << " + ";
            temp = temp->next;
        }
        cout << endl;
    }

    Polynomial add(Polynomial &other) {
        Polynomial result;
        Term *p1 = head, *p2 = other.head;
        while (p1 != NULL || p2 != NULL) {
            if (p1 == NULL) { result.insertTerm(p2->coeff, p2->exp); p2 = p2->next; }
            else if (p2 == NULL) { result.insertTerm(p1->coeff, p1->exp); p1 = p1->next; }
            else if (p1->exp == p2->exp) {
                result.insertTerm(p1->coeff + p2->coeff, p1->exp);
                p1 = p1->next; p2 = p2->next;
            }
            else if (p1->exp > p2->exp) { result.insertTerm(p1->coeff, p1->exp); p1 = p1->next; }
            else { result.insertTerm(p2->coeff, p2->exp); p2 = p2->next; }
        }
        return result;
    }
};

int main() {
    Polynomial p1, p2;
    int n, coeff, exp;

    cout << "Enter number of terms in first polynomial: "; cin >> n;
    cout << "Enter coeff and exp for each term:\n";
    for (int i = 0; i < n; i++) { cin >> coeff >> exp; p1.insertTerm(coeff, exp); }

    cout << "Enter number of terms in second polynomial: "; cin >> n;
    cout << "Enter coeff and exp for each term:\n";
    for (int i = 0; i < n; i++) { cin >> coeff >> exp; p2.insertTerm(coeff, exp); }

    cout << "\nPolynomial 1: "; p1.display();
    cout << "Polynomial 2: "; p2.display();

    Polynomial p3 = p1.add(p2);
    cout << "Sum: "; p3.display();
    return 0;
}

\section*{Program 28: Fibonacci Series using Recursion}
\addcontentsline{toc}{section}{Program 28: Fibonacci Series}

\subsection*{Problem Statement}
Write a program to display the Fibonacci series using recursion.

\subsection*{Algorithm}
\begin{enumerate}
    \item Base case: If n == 0, return 0. If n == 1, return 1.
    \item Recursive case: Return fib(n-1) + fib(n-2).
\end{enumerate}

\subsection*{Source Code (28.cpp)}
\begin{lstlisting}[style=cppstyle]
#include <iostream>
using namespace std;

int fib(int n) {
    if (n == 0) return 0;
    if (n == 1) return 1;
    return fib(n - 1) + fib(n - 2);
}

int main() {
    int n;
    cout << "Enter number of terms: ";
    cin >> n;

    cout << "Fibonacci Series: ";
    for (int i = 0; i < n; i++)
        cout << fib(i) << " ";
    cout << endl;

    return 0;
}
\end{lstlisting}
